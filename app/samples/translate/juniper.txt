# 接入交换机 SW-ACC-06（Junos EX，set 形式）—— 整机配置小样
set system host-name SW-ACC-06
set system root-authentication encrypted-password "$6$rounds=10000$abcdefgh"
set vlans VLAN10-OFFICE vlan-id 10
set vlans VLAN20-FINANCE vlan-id 20
set interfaces ge-0/0/1 description "TO-PC-AREA-A"
set interfaces ge-0/0/1 unit 0 family ethernet-switching interface-mode access
set interfaces ge-0/0/1 unit 0 family ethernet-switching vlan members VLAN10-OFFICE
set interfaces ge-0/0/1 unit 0 family ethernet-switching interface-mac-limit 2 packet-action drop
set interfaces ge-0/0/2 description "TO-PC-AREA-B"
set interfaces ge-0/0/2 unit 0 family ethernet-switching interface-mode access
set interfaces ge-0/0/2 unit 0 family ethernet-switching vlan members VLAN20-FINANCE
set interfaces ge-0/0/24 description "TO-CORE-SW1"
set interfaces ge-0/0/24 unit 0 family ethernet-switching interface-mode trunk
set interfaces ge-0/0/24 unit 0 family ethernet-switching vlan members VLAN10-OFFICE
set interfaces ge-0/0/24 unit 0 family ethernet-switching vlan members VLAN20-FINANCE
set interfaces ge-0/0/24 unit 0 family ethernet-switching native-vlan-id 1
set chassis aggregated-devices ethernet device-count 2
set interfaces ae0 aggregated-ether-options lacp active
set interfaces ae0 unit 0 family ethernet-switching interface-mode trunk
set interfaces ae0 unit 0 family ethernet-switching vlan members VLAN10-OFFICE
set interfaces ge-0/0/3 ether-options 802.3ad ae0
set interfaces ge-0/0/4 ether-options 802.3ad ae0
set routing-options router-id 6.6.6.6
set protocols ospf area 0.0.0.0 interface ge-0/0/24.0
set routing-options static route 192.168.20.0/24 next-hop 10.0.0.2
set system ntp server 10.1.1.10
set protocols rstp interface all
# ---- V 批扩族样例：PoE / 风暴抑制 ----
set poe interface ge-0/0/5
set storm-control-profiles SC-ACCESS all bandwidth-percentage 5
set interfaces ge-0/0/5 unit 0 family ethernet-switching storm-control SC-ACCESS
# ---- V 批扩族样例：系统管理（AAA / DNS / SSH / SNMP / 保存 / 显示）----
set system radius-server 10.1.1.5
set system authentication-order radius
set system authentication-order password
set system name-server 223.5.5.5
set system services ssh protocol-version v2
set system services ssh root-login deny
set snmp location "IDC-A-机柜12"
set snmp contact "网络运维组"
set snmp trap-group NMS-TRAP targets 10.0.0.100
save /var/tmp/SW-ACC-06-20261106
request system configuration rescue save
set cli screen-length 0
# ---- 运维查询行（迁移后现场核对用，非 running-config 内容）----
show version
show bgp summary
show virtual-chassis status
show spanning-tree bridge
show log messages | last 50
