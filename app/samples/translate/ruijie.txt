# 接入交换机 SW-ACC-04（锐捷 RGOS）—— 整机配置小样
hostname SW-ACC-04
!
vlan 10
 name OFFICE
!
vlan 20
 name FINANCE
!
interface GigabitEthernet0/1
 description TO-PC-AREA-A
 switchport mode access
 switchport access vlan 10
 switchport port-security
 switchport port-security maximum 2
 switchport port-security violation protect
!
interface GigabitEthernet0/2
 description TO-PC-AREA-B
 switchport mode access
 switchport access vlan 20
!
interface GigabitEthernet0/24
 description TO-CORE-SW1
 switchport mode trunk
 switchport trunk allowed vlan add 10,20
 switchport trunk native vlan 1
!
interface aggregateport 1
 switchport mode trunk
!
interface range GigabitEthernet0/2-3
 port-group 1 mode active
!
router ospf 1
 router-id 4.4.4.4
 network 10.0.0.0 0.0.0.255 area 0
 passive-interface GigabitEthernet0/24
!
ip route 192.168.20.0 255.255.255.0 10.0.0.2
!
ntp server 10.1.1.10
!
# ---- V 批扩族样例：ACL / NAT / VRRP / PoE / 风暴抑制 ----
ip access-list extended ACL-OFFICE-02
 permit ip 192.168.10.0 0.0.0.255 10.10.0.0 0.0.255.255
 deny ip any any
!
interface GigabitEthernet0/5
 description TO-PC-AREA-C
 switchport mode access
 switchport access vlan 10
 ip access-group ACL-OFFICE-02 in
 poe enable
 storm-control broadcast level 5
!
access-list 1 permit 192.168.10.0 0.0.0.255
ip nat inside source list 1 interface GigabitEthernet0/24 overload
!
interface GigabitEthernet0/6
 description TO-LAN-NAT
 ip nat inside
!
interface Vlan10
 ip address 10.10.10.1 255.255.255.0
 vrrp 10 ip 10.10.10.254
 vrrp 10 priority 120
!
# ---- V 批扩族样例：系统管理（AAA / DNS / SSH / Console / SNMP / 保存 / 回滚 / 显示）----
radius-server host 10.1.1.5
aaa authentication login default radius local
!
ip name-server 223.5.5.5
!
ip domain-name corp.local
ip ssh version 2
line vty 0 4
 login local
 transport input ssh
!
line console 0
 logging synchronous
 exec-timeout 10 0
 login
!
snmp-server host 10.0.0.100 version 2c RO@Public1
snmp-server location IDC-A-机柜12
snmp-server contact 网络运维组
!
copy running-config flash:backup-20261106.cfg
copy flash:/backup-20261106.cfg running-config
terminal length 0
!
# ---- 运维查询行（迁移后现场核对用，非 running-config 内容）----
show version
show ip bgp summary
show switch virtual
show spanning-tree summary
show logging
!
end
